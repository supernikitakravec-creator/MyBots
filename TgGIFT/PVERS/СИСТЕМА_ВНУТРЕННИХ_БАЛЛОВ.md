# 💰 СИСТЕМА ВНУТРЕННИХ БАЛЛОВ ДЛЯ TGGIFT BOT

## 🎯 Концепция Системы

### Почему внутренние баллы?
1. **Обход ограничений** - Telegram Stars нельзя передавать между пользователями
2. **Универсальность** - Один баланс для всех способов оплаты (ЮКасса, TON, др.)
3. **Гибкость** - Можем менять курсы, делать акции, бонусы
4. **Контроль** - Полный контроль над экономикой бота

### Базовая схема:
```
Пользователь → Пополнение (₽/TON) → Конвертация в баллы → Использование в боте
```

## 💎 Структура Баллов

### Курсы конвертации:
```python
CONVERSION_RATES = {
    "RUB": 1.0,      # 1 рубль = 1 балл
    "TON": 350.0,    # 1 TON = 350 баллов (примерный курс)
    "USD": 90.0,     # 1 USD = 90 баллов
}

# Бонусы при пополнении
TOPUP_BONUSES = {
    1000: 0,      # < 1000 баллов - без бонуса
    5000: 0.05,   # 1000-5000 баллов - бонус 5%
    10000: 0.10,  # 5000-10000 баллов - бонус 10%
    50000: 0.15,  # > 10000 баллов - бонус 15%
}
```

```python
# Добавляем новые методы в DatabaseManager

def init_points_system(self):
    """Инициализация системы баллов"""
    with self.get_connection() as conn:
        cursor = conn.cursor()
        
        # Создаем таблицы для системы баллов
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS internal_balances (
                user_id INTEGER PRIMARY KEY,
                balance_points INTEGER DEFAULT 0,
                total_earned INTEGER DEFAULT 0,
                total_spent INTEGER DEFAULT 0,
                last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS points_transactions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER NOT NULL,
                type TEXT NOT NULL,
                amount INTEGER NOT NULL,
                balance_after INTEGER NOT NULL,
                source TEXT,
                description TEXT,
                metadata TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (user_id) REFERENCES users(user_id)
            )
        ''')
        
        conn.commit()

def get_points_balance(self, user_id: int) -> int:
    """Получение баланса в баллах"""
    try:
        with self.get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                'SELECT balance_points FROM internal_balances WHERE user_id = ?',
                (user_id,)
            )
            row = cursor.fetchone()
            return row[0] if row else 0
    except Exception as e:
        logger.error(f"Ошибка получения баланса баллов: {e}")
        return 0

def add_points(self, user_id: int, amount: int, source: str, 
               description: str = "", metadata: dict = None) -> bool:
    """Добавление баллов пользователю"""
    if amount <= 0:
        return False
    
    try:
        with self.get_connection() as conn:
            cursor = conn.cursor()
            
            # Получаем текущий баланс
            current_balance = self.get_points_balance(user_id)
            new_balance = current_balance + amount
            
            # Обновляем или создаем запись баланса
            cursor.execute('''
                INSERT INTO internal_balances (user_id, balance_points, total_earned)
                VALUES (?, ?, ?)
                ON CONFLICT(user_id) DO UPDATE SET
                    balance_points = balance_points + ?,
                    total_earned = total_earned + ?,
                    last_updated = CURRENT_TIMESTAMP
            ''', (user_id, amount, amount, amount, amount))
            
            # Записываем транзакцию
            cursor.execute('''
                INSERT INTO points_transactions 
                (user_id, type, amount, balance_after, source, description, metadata)
                VALUES (?, 'topup', ?, ?, ?, ?, ?)
            ''', (user_id, amount, new_balance, source, description, 
                  json.dumps(metadata) if metadata else None))
            
            conn.commit()
            logger.info(f"Добавлено {amount} баллов пользователю {user_id}")
            return True
            
    except Exception as e:
        logger.error(f"Ошибка добавления баллов: {e}")
        return False

def spend_points(self, user_id: int, amount: int, 
                 description: str = "") -> bool:
    """Списание баллов"""
    if amount <= 0:
        return False
    
    try:
        with self.get_connection() as conn:
            cursor = conn.cursor()
            
            # Проверяем баланс
            current_balance = self.get_points_balance(user_id)
            if current_balance < amount:
                return False
            
            new_balance = current_balance - amount
            
            # Обновляем баланс
            cursor.execute('''
                UPDATE internal_balances 
                SET balance_points = ?,
                    total_spent = total_spent + ?,
                    last_updated = CURRENT_TIMESTAMP
                WHERE user_id = ?
            ''', (new_balance, amount, user_id))
            
            # Записываем транзакцию
            cursor.execute('''
                INSERT INTO points_transactions 
                (user_id, type, amount, balance_after, source, description)
                VALUES (?, 'spend', ?, ?, 'internal', ?)
            ''', (user_id, -amount, new_balance, description))
            
            conn.commit()
            return True
            
    except Exception as e:
        logger.error(f"Ошибка списания баллов: {e}")
        return False

def convert_payment_to_points(self, user_id: int, amount: float, 
                            currency: str, payment_id: str) -> dict:
    """Конвертация платежа в баллы с учетом бонусов"""
    try:
        # Получаем курс конвертации
        rate = CONVERSION_RATES.get(currency, 1.0)
        base_points = int(amount * rate)
        
        # Вычисляем бонус
        bonus_percent = 0
        for threshold, bonus in sorted(TOPUP_BONUSES.items()):
            if base_points >= threshold:
                bonus_percent = bonus
        
        bonus_points = int(base_points * bonus_percent)
        total_points = base_points + bonus_points
        
        # Сохраняем конвертацию
        with self.get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute('''
                INSERT INTO conversion_history 
                (user_id, source_currency, source_amount, points_amount, 
                 conversion_rate, bonus_percent, bonus_points, payment_id)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            ''', (user_id, currency, amount, total_points, rate, 
                  bonus_percent, bonus_points, payment_id))
            conn.commit()
        
        # Добавляем баллы
        self.add_points(
            user_id, 
            total_points, 
            currency.lower(),
            f"Пополнение через {currency}",
            {"payment_id": payment_id, "bonus": bonus_points}
        )
        
        return {
            "success": True,
            "base_points": base_points,
            "bonus_points": bonus_points,
            "total_points": total_points,
            "bonus_percent": bonus_percent * 100
        }
        
    except Exception as e:
        logger.error(f"Ошибка конвертации: {e}")
        return {"success": False, "error": str(e)}
```

## 💳 Интеграция с ЮКассой

```python
# payment_handler.py
class PaymentHandler:
    def __init__(self, db_manager, yookassa_provider, ton_provider):
        self.db = db_manager
        self.yookassa = yookassa_provider
        self.ton = ton_provider
    
    async def handle_yookassa_webhook(self, payment_data: dict):
        """Обработка webhook от ЮКассы"""
        if payment_data['status'] == 'succeeded':
            user_id = int(payment_data['metadata']['user_id'])
            amount = float(payment_data['amount']['value'])
            payment_id = payment_data['id']
            
            # Конвертируем в баллы
            result = self.db.convert_payment_to_points(
                user_id, amount, 'RUB', payment_id
            )
            
            if result['success']:
                # Отправляем уведомление пользователю
                await self.notify_user_topup(
                    user_id, 
                    amount,
                    result['total_points'],
                    result['bonus_points']
                )
    
    async def handle_ton_payment(self, user_id: int, amount_ton: float, tx_hash: str):
        """Обработка TON платежа"""
        result = self.db.convert_payment_to_points(
            user_id, amount_ton, 'TON', tx_hash
        )
        
        if result['success']:
            await self.notify_user_topup(
                user_id,
                f"{amount_ton} TON",
                result['total_points'],
                result['bonus_points']
            )
```

```python
async def buy_subscription_points(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Покупка подписки за баллы"""
    query = update.callback_query
    user_id = query.from_user.id
    sub_type = query.data.split('_')[-1]  # basic или vip
    
    # Получаем цену
    price = POINTS_PRICES['subscription'][sub_type]
    
    # Проверяем баланс
    balance = db_manager.get_points_balance(user_id)
    
    if balance < price:
        await query.answer(
            f"❌ Недостаточно баллов! Нужно: {price}, у вас: {balance}",
            show_alert=True
        )
        return
    
    # Списываем баллы
    if db_manager.spend_points(user_id, price, f"Подписка {sub_type}"):
        # Активируем подписку
        db_manager.activate_subscription(user_id, sub_type)
        
        await query.message.edit_text(
            f"✅ <b>Подписка {sub_type.upper()} активирована!</b>\n\n"
            f"💎 Списано: {price} баллов\n"
            f"📅 Действует: 30 дней\n\n"
            f"Спасибо за покупку! 🎉",
            parse_mode='HTML'
        )
    else:
        await query.answer("❌ Ошибка при покупке", show_alert=True)
```
